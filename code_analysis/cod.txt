        String query = "select * from employee_performance order by performanceScore desc";

        while(rs.next()){
            int id = rs.getInt(1);
            String name = rs.getString(2);
            String dep = rs.getString(3);
            int sal = rs.getInt(4);
            String stat = rs.getString(5);
            int score = rs.getInt(6);

    public static void insert(Connection con, Statement stmt, Scanner sc) throws SQLException{
        
        PreparedStatement ptmt = null;
      
        int id = sc.nextInt();
        sc.nextLine();
        String name = sc.nextLine();
        String dep = sc.nextLine();
        int sal = sc.nextInt();
        sc.nextLine();
        String stat = sc.nextLine();
        int score = sc.nextInt();
        
        if(sal>7000){
            String query = "insert into employee_performance values(?,?,?,?,?,?)";
            ptmt = con.prepareStatement(query);
            
            ptmt.setInt(1, id);
            ptmt.setString(2, name);
            ptmt.setString(3, dep);
            ptmt.setInt(4, sal);
            ptmt.setString(5, stat);
            ptmt.setInt(6, score);

                System.out.println("Employee record added successfully.");
                sort(con, stmt, sc);

    public static void update(Connection con, Statement stmt, Scanner sc) throws SQLException{
        String query = "Update employee_performance set projectStatus=?, performanceScore=? where projectStatus='In progress' and performanceScore<70";
        
        String stat = sc.nextLine();
        int score = sc.nextInt();

        PreparedStatement ptmt = con.prepareStatement(query);
        
        ptmt.setString(1, stat);
